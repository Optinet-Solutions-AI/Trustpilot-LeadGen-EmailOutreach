playwright>=1.40.0
python-dotenv>=1.0.0
postgrest>=0.10.6,<0.21
playwright-stealth>=1.0.6
requests>=2.31.0
# curl_cffi — Chrome-TLS-fingerprint HTTP for the Tier 1.5 enricher fallback.
# Pre-built wheels are available for Windows / macOS / manylinux, so pip install
# does not need a C toolchain in normal environments.
curl_cffi>=0.7.0

# BeautifulSoup + lxml — HTML parsing for the TripAdvisor and Yelp platform
# plugins. Both import bs4 at module-top, so the package MUST be available
# wherever tools.scraper.platforms is imported (Cloud Run for taxonomy
# refresh; EC2 worker for actual scraping). Discovered missing on Cloud Run
# 2026-05-18 when GET /api/scrape/taxonomy/refresh?platform=yelp returned
# `ModuleNotFoundError: No module named 'bs4'` from the Python subprocess
# spawned by server/src/services/taxonomy-discovery.ts.
beautifulsoup4>=4.12.0
lxml>=5.0.0

# cryptography — AES-256-GCM for social_accounts.encrypted_cookies.
# Required by tools/scraper/shared/encryption.py; cookies must round-trip
# byte-compatibly with server/src/lib/encryption.ts (Node built-in crypto).
# Pinned <42 because pyOpenSSL 23.3.x (which we hold to keep selenium-wire's
# bundled mitmproxy working — see pyOpenSSL pin below) requires
# cryptography>=41.0.5,<42 in its metadata. AESGCM has been stable since
# cryptography 38, so 41.x is functionally identical for our use.
cryptography>=41.0.5,<42

# undetected-chromedriver — anti-fingerprint Chrome driver for Facebook/
# Instagram. Required by tools/scraper/shared/login_flows.py (operator
# login) and tools/scraper/platforms/{facebook,instagram}.py (actual
# scraping sessions). Playwright's CDP signals are detected by Meta's
# fingerprinting; this driver patches them out.
#
# setuptools<81 is a transitive workaround: undetected-chromedriver 3.5.x
# still imports the stdlib `distutils` module, which Python 3.12+ removed.
# setuptools<81 reinstates `distutils` as a compat shim. Setuptools 81+
# dropped that shim, so pin below 81 until undetected-chromedriver 4.x
# (which doesn't depend on distutils) lands.
setuptools<81
undetected-chromedriver>=3.5.5
selenium>=4.18.0

# selenium-wire — runs a local intercepting proxy that handles upstream
# proxy authentication in Python instead of relying on a Chrome Manifest
# V2 extension. Required when routing FB scrapes through a residential
# proxy (RESIDENTIAL_PROXY_* env vars in facebook.py): Chrome 128+
# silently disables V2 extensions in --headless=new mode, breaking the
# webRequest.onAuthRequired auth-injection pattern, so the connection
# returns 407 and Chrome renders an empty body. selenium-wire bypasses
# that entirely by accepting Proxy-Authorization in its Python proxy.
# undetected-chromedriver auto-detects the `seleniumwire_options` kwarg
# and swaps to selenium-wire's driver subclass — no other code changes.
selenium-wire>=5.1.0
# blinker pin: selenium-wire 5.1 imports blinker.base.WeakNamespace which
# was removed in blinker 1.8. Pin until selenium-wire updates upstream.
blinker<1.8
# pyOpenSSL pin: selenium-wire 5.1's bundled mitmproxy still calls
# X509.get_extension(i), which pyOpenSSL 24.0 deprecated and 26.x
# REMOVED. Without this pin, every TLS handshake through selenium-wire's
# local interceptor crashes with `AttributeError: 'X509' object has no
# attribute 'get_extension'`. 23.3.0 is the last version that exposes
# the old API. Re-evaluate when selenium-wire ships a newer mitmproxy.
pyOpenSSL>=23.3.0,<24.0
